package com.devops;

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

class AppTest {

    @Test
    void testGreeting() {
        App app = new App();

        assertEquals(
            "Hello, Praveen!",
            app.greet("Praveen")
        );
    }

    @Test
    void testGuestGreeting() {
        App app = new App();

        assertEquals(
            "Hello, Guest!",
            app.greet("")
        );
    }
}
